import requests
URL = "{{ ping_url }}"

def do_work():
    # 在此处进行数据处理、备份转储、邮件发送等工作。
    # 成功时返回真值。
    # 失败时返回假值或抛出异常。
    return True

success = False
try:
    success = do_work()
finally:
    # 成功时，请求 {{ ping_url }}
    # 失败时，请求 {{ ping_url }}/fail
    requests.get(URL if success else URL + "/fail")
